# Виды кабельных линий (способы прокладки) «СЛТ Графика».
# Свои виды и правка встроенных — SLT_LINES (файл %APPDATA%\SltGrafika\lines.txt): вид с тем же id заменяет встроенный.
#
# = id | Наименование | цвет ACI (0 — по слою) | t — толстая, пусто — тонкая | штрихи мм (штрих пробел …), пусто — сплошная | шаг знака, мм (0 — без знака)
# Ниже — знак, повторяемый вдоль линии (мм листа; x — вдоль линии, y — поперёк, влево): команды библиотеки L, P, C, A, F, D, T.

= open | Открыто | 0 | t | | 0

= pipe | В гладкой трубе | 0 | t | | 25
C 0 1.9 0.7

= corr | В гофрированной трубе | 0 | t | | 25
C 0 1.9 0.8
C 0 1.9 0.35

= tray | В лотке | 0 | t | | 30
P -1.5 1.2 -1.5 -1.3 1.5 -1.3 1.5 1.2

= box | В коробе | 0 | t | | 30
P -1.5 -1 1.5 -1 1.5 1 -1.5 1 c

= hose | В металлорукаве | 0 | t | | 20
A -2.4 0 1.2 0 180
A 0 0 1.2 180 360
